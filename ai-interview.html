function hasAny(s,sub){ return sub && s.includes(sub.trim()); }

function splitWords(s){ return (s||"").split(/[·,，、\s/]+/).filter(w=>w.length>=2); }

function containsAny(s,words){ return words.some(w=>s.includes(w)); }

  recog.onstart=()=>{ listening=true; micBtn.classList.add("rec"); micBtn.textContent="⏹"; voiceStatus.textContent="🎙️ 正在聆听，请开始回答…"; };

  recog.onend=()=>{ listening=false; micBtn.classList.remove("rec"); micBtn.textContent="🎤"; if(voiceStatus.textContent.includes("聆听")) voiceStatus.textContent=""; };

  recog.onerror=(e)=>{ if(e.error==="not-allowed"){ voiceStatus.textContent="⚠️ 无法访问麦克风，请在浏览器设置中允许麦克风权限（或改用键盘输入）。"; } else if(e.error==="no-speech"){ voiceStatus.textContent="⚠️ 未检测到语音，请再试一次。"; } };

  recog.onresult=(e)=>{ let interim="";

    for(let i=e.resultIndex;i<e.results.length;i++){ const t=e.results[i][0].transcript;

      if(e.results[i].isFinal){ baseText=(baseText?baseText:"")+t; } else interim+=t; }

    document.getElementById("userInput").value=(baseText+interim).trim(); };

  micBtn.onclick=()=>{ if(listening){ recog.stop(); return; }

    baseText=document.getElementById("userInput").value?document.getElementById("userInput").value+" ":"";

    try{ recog.start(); }catch(err){} };

}else{

  micBtn.disabled=true; micBtn.textContent="🚫"; micBtn.title="当前浏览器不支持语音识别";

  voiceStatus.textContent="⚠️ 当前浏览器不支持语音识别，请使用 Chrome/Edge，或改用键盘输入。";

}

function speak(text){

  if(!document.getElementById("ttsToggle").checked||!window.speechSynthesis) return;

  window.speechSynthesis.cancel();

  const plain=text.replace(/<[^>]+>/g,"");

  const u=new SpeechSynthesisUtterance(plain); u.lang="zh-CN"; u.rate=1;

  window.speechSynthesis.speak(u);

}
